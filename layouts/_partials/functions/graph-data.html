{{/* Graph Data Function Returns the graph data (pages and link relationships) Usage: {{ partial "functions/graph-data" "graph-data-key" }} */}} {{- $pages := site.RegularPages -}} {{- $pairs := partialCached "functions/backlink-index" "backlink-index-key" -}} {{/* Build page dictionary */}} {{- $ps := newScratch -}} {{- range $pages -}} {{- $title := partial "functions/get-title" (dict "page" .) -}} {{- $ps.SetInMap "pages" .RelPermalink (dict "permalink" .RelPermalink "title" ($title | plainify | htmlUnescape | chomp | replaceRE `\s+` " " | title) "section" .Section ) -}} {{- end -}} {{/* Build graph relationships from the site-wide relation index (functions/backlink-index.html). That index understands both Markdown links [text](/path) and wikilinks [[note]] — the parser that used to live here only matched `](/path)`, so once content moved to wikilinks this graph silently lost every edge. Reusing the index also drops the per-link site.GetPage scanning and the 1500-entry dict merging that made this partial expensive. */}} {{- $gs := newScratch -}} {{- range $pages -}} {{- $gs.SetInMap "graph" .RelPermalink (dict "in" slice "out" slice) -}} {{- end -}} {{- range $pairs -}} {{- $s := .s -}} {{- $t := .t -}} {{- $graph := $gs.Get "graph" -}} {{- $src := index $graph $s -}} {{- $dst := index $graph $t -}} {{- if $src -}} {{- if not (in $src.out $t) -}} {{- $gs.SetInMap "graph" $s (dict "in" $src.in "out" ($src.out | append $t)) -}} {{- end -}} {{- end -}} {{- if $dst -}} {{- if not (in $dst.in $s) -}} {{- $gs.SetInMap "graph" $t (dict "in" ($dst.in | append $s) "out" $dst.out) -}} {{- end -}} {{- end -}} {{- end -}} {{- return (dict "pages" ($ps.Get "pages") "graph" ($gs.Get "graph")) }}
