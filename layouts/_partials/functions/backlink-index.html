{{/* Site-wide backlink index — built once per site build (partialCached). Walks the raw Markdown of every content page and records internal references: {s: source permalink, t: target permalink, c: context line} Scans site.RegularPages, not just posts+til: root-level pages (start.md, watch.md, …) link to notes too, and those backlinks belong in the target's "引用这篇的" list. The context line is the sentence the link appears in, stripped of Markdown syntax and truncated — it renders as the "linked mention" under each backlink (the Obsidian/Influx pattern), which is what makes a backlink list readable rather than a bare list of titles. Both link styles are indexed: - standard Markdown [text](/path) - wikilinks [[Note name]] / [[Note name|Label]] Scanning line by line (instead of findRE over the whole file) is what makes the context cheap; unresolved targets are dropped here and reported by layouts/_markup/render-link.html and content/render-content.html instead. Note: the naive Hugo backlink partials in the wild are O(n²) — one full-site scan per rendered page. This is one scan per build, cached. */}} {{- $pages := site.RegularPages -}} {{- $wlm := partialCached "functions/wikilink-map" "wikilink-map-key" -}} {{- $pairs := slice -}} {{- range $pages -}} {{- $src := .RelPermalink -}} {{- range $line := split .RawContent "\n" -}} {{- $ctx := $line -}} {{- $ctx = replaceRE `\[\[[^\]\[]*\|([^\]\[]*)\]\]` "$1" $ctx -}} {{- $ctx = replaceRE `\[\[([^\]\[]+)\]\]` "$1" $ctx -}} {{- $ctx = replaceRE `\[([^\]]*)\]\([^)]*\)` "$1" $ctx -}} {{- $ctx = replaceRE "`" "" $ctx -}} {{- $ctx = replaceRE `[*_#>\s]+` " " $ctx -}} {{- $ctx = trim $ctx " " -}} {{- $ctx = $ctx | truncate 140 -}} {{/* standard Markdown links */}} {{- range findRE `\]\((/[^)\s]+)\)` $line -}} {{- $dest := strings.TrimSuffix ")" (strings.TrimPrefix "](" .) -}} {{- $path := $dest -}} {{- if in $path "#" -}}{{- $path = index (split $path "#") 0 -}}{{- end -}} {{- with site.GetPage $path -}} {{- $pairs = $pairs | append (dict "s" $src "t" .RelPermalink "c" $ctx) -}} {{- end -}} {{- end -}} {{/* wikilinks */}} {{- range findRE `\[\[[^\]\[]+\]\]` $line -}} {{- $inner := strings.TrimPrefix "[[" (strings.TrimSuffix "]]" .) -}} {{- $target := index (split $inner "|") 0 -}} {{- /* 表格单元格里写成 [[x\|别名]]，RawContent 里带着转义反斜杠，去掉再查 */ -}} {{- $target = strings.Replace $target "\\" "" -}} {{- $target = trim $target " " -}} {{- if in $target "#" -}}{{- $target = index (split $target "#") 0 -}}{{- end -}} {{- $href := index $wlm (lower $target) -}} {{- if not $href -}} {{- $href = index $wlm (lower (replace $target " " "-")) -}} {{- end -}} {{- with $href -}} {{- $pairs = $pairs | append (dict "s" $src "t" . "c" $ctx) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $pairs -}}
