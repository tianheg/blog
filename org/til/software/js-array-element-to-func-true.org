原始代码：

#+BEGIN_SRC js
function findElement(arr, func) {
let num = 0
return num
}

console.log(findElement([1, 2, 3, 4], (num) => num % 2 === 0))
#+END_SRC

看过参考答案后，发现：原来还可以这样！

** for

#+BEGIN_SRC js
function findElement(arr, func) {
let num = 0

for (let i = 0; i < arr.length; i++) {
num = arr[i]
if (func(num)) { return num }
}
return undefined
}

console.log(findElement([1, 2, 3, 4], (num) => num % 2 === 0))
#+END_SRC

** find()

#+BEGIN_SRC js
function findElement(arr, func) { return arr.find(func) }

console.log(findElement([1, 2, 3, 4], (num) => num % 2 === 0))
#+END_SRC

** map()

#+BEGIN_SRC js
function findElement(arr, func) { return arr[arr.map(func).indexOf(true)] }

console.log(findElement([1, 2, 3, 4], (num) => num % 2 === 0))
#+END_SRC

** 递归

#+BEGIN_SRC js
function findElement(arr, func) { return arr.length && !func(arr[0]) ? findElement(arr.slice(1), func) : arr[0] }

console.log(findElement([1, 2, 3, 4], (num) => num % 2 === 2))
#+END_SRC
